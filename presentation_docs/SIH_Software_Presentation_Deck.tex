\section{Slide 2: Problem Statement — The Low Earth Orbit Crisis}

\subsection*{The Modern Orbital Environment}
\begin{itemize}[leftmargin=1.5em]
    \item Over \textbf{27,000+} cataloged artificial objects larger than 10~cm orbit Earth at velocities exceeding \textbf{7.5~km/s (27,000~km/h)}.
    \item With commercial mega-constellations (Starlink, OneWeb, Project Kuiper), satellite density in Low Earth Orbit (LEO) has surged by over \textbf{400\%} in five years.
\end{itemize}

\subsection*{Real-World Case Study: The 10 February 2009 Collision}
\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=\textbf{Historical Ground Truth Benchmark}]
On \textbf{10 February 2009 at 16:56 UTC}, at an altitude of \textbf{789~km above the Taymyr Peninsula, Siberia}:
\begin{itemize}
    \item \textbf{Iridium 33} (an active 560~kg commercial communications satellite) collided with
    \item \textbf{Cosmos 2251} (a derelict 900~kg Russian military satellite)
    \item \textbf{Relative Collision Velocity}: \textbf{14.12~km/s (50,832~km/h)} — a near head-on intersection.
    \item \textbf{Consequence}: Both spacecraft were obliterated in milliseconds, generating over \textbf{2,000 trackable debris fragments} and thousands of untrackable lethal shards that continue to threaten the ISS and operational satellites today (Kessler Syndrome).
\end{itemize}
\end{tcolorbox}

\subsection*{Why Existing Screening Approaches Fail}
\begin{enumerate}[leftmargin=1.5em]
    \item \textbf{Computational Bottleneck}: Direct pairwise checking of $N$ satellites requires $O(N^2)$ calculations, causing severe latency in operational pipelines.
    \item \textbf{False-Alarm Fatigue}: Naive distance-only screening produces thousands of daily warnings without realistic collision probabilities ($P_c$), overwhelming operators.
    \item \textbf{Manual Maneuver Delays}: Human-in-the-loop maneuver calculation takes hours when critical orbital decisions must be executed in minutes.
\end{enumerate}

\section{Slide 4: Implementation — Physics, Math \& Computer Science}

\subsection{1. Physics and Mathematics}

\subsubsection*{A. SGP4 Orbital Propagation}
Converts Two-Line Element (TLE) mean Keplerian orbital elements ($a, e, i, \Omega, \omega, M$) into instantaneous Cartesian state vectors $[\mathbf{r}, \mathbf{v}]$ in the Geocentric Celestial Reference System (GCRS/ECI non-rotating frame). SGP4 models non-spherical Earth gravitational harmonics ($J_2, J_3, J_4$) and atmospheric drag ($B^*$).

\subsubsection*{B. Two-Stage Conjunction Screening}
\begin{itemize}
    \item \textbf{Stage 1 (Coarse Filter)}: Filters all candidate pairs by checking overlap of apogee and perigee altitudes with a safety buffer:
    $$r_{\text{perigee}, i} - 50\text{ km} \le r_{\text{apogee}, j} \quad \text{and} \quad r_{\text{perigee}, j} - 50\text{ km} \le r_{\text{apogee}, i}$$
    This discards $>90\%$ of non-threatening pairs in milliseconds.
    \item \textbf{Stage 2 (Fine TCA Search)}: Applies \texttt{scipy.optimize.minimize\_scalar} (Golden-Section / Brent minimization) to the scalar relative distance function:
    $$f(t) = \|\mathbf{r}_i(t) - \mathbf{r}_j(t)\|$$
    pinpointing the exact Time of Closest Approach ($t_{\text{TCA}}$).
\end{itemize}

\subsubsection*{C. Foster/Alfano 2D Gaussian Probability ($P_c$)}
In the encounter B-plane normal to relative velocity $\mathbf{v}_{\text{rel}}$, the collision integral evaluates the 2D Gaussian probability density over the hard-body circle of combined radius $\text{HBR} = 10\text{ m}$:
\begin{equation}
P_c \approx \frac{\text{HBR}^2}{2\sigma^2} \exp\left(-\frac{d_{\text{miss}}^2}{2\sigma^2}\right)
\end{equation}
Because $\text{HBR} \ll \sigma$ ($\sigma = 500\text{ m}$ assumed isotropic TLE uncertainty), probability density is uniform across the disk, yielding exact analytical computation in $<10\ \mu\text{s}$.

\subsubsection*{D. Clohessy-Wiltshire (CW) Maneuver Optimization}
Linearized relative motion in near-circular orbit ($e < 0.01$) is governed by the state transition matrix:
$$\mathbf{r}(\Delta t) = \Phi_{rr}(\Delta t)\mathbf{r}_0 + \Phi_{rv}(\Delta t)(\mathbf{v}_0 + \Delta\mathbf{v})$$
The optimal burn direction is the right singular vector $\mathbf{v}_1$ from the Singular Value Decomposition of $\Phi_{rv}$:
$$\Phi_{rv}(\Delta t) = U \Sigma V^T \implies \Delta\mathbf{v}_{\text{opt}} = \Delta V \cdot V_{[:, 0]}$$
Along-track displacement accumulates via the secular term $\Delta y \approx -3 n \Delta t \Delta v_y$, providing \textbf{14$\times$ more separation} at $\Delta t = 24\text{h}$ than at $\Delta t = 1\text{h}$.

\subsection{2. Computer Science Architecture}
\begin{itemize}
    \item \textbf{Backend Microservices}: Asynchronous Python 3.12+ FastAPI server utilizing vectorized NumPy arrays and SciPy numerical optimizers.
    \item \textbf{Frontend Telemetry Workstation}: React 19 with Vite, Tailwind CSS, and Three.js WebGL canvas rendering 60 FPS 3D orbital dynamics.
    \item \textbf{Surrogate Acceleration Layer}: Pre-trained Random Forest classifier acting as an acceleration surrogate to pre-score pairs before deep propagation.
\end{itemize}

\subsection{3. Satellite Data Acquisition Pipeline}
\begin{itemize}
    \item Direct automated fetch from \textbf{CelesTrak / Space-Track} active satellite catalogs.
    \item Local persistent cache with automated Time-To-Live (TTL) expiration to minimize external API rate limits.
\end{itemize}
